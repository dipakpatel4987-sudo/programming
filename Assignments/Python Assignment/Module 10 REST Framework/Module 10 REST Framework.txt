Practical Example 1: Fetch a random joke from an API and print it

# joke.py   (run: pip install requests, then python joke.py)
import requests

def get_joke():
    url = 'https://official-joke-api.appspot.com/random_joke'
    try:
        response = requests.get(url, timeout=10)
        response.raise_for_status()               # raises an error for 4xx or 5xx codes
        joke = response.json()                    # JSON text -> Python dictionary
        print('Setup    :', joke['setup'])
        print('Punchline:', joke['punchline'])
    except requests.exceptions.RequestException as error:
        print('Could not fetch a joke:', error)

if __name__ == '__main__':
    get_joke()
The script sends an HTTP GET request, checks the status code, converts the JSON reply to a dictionary, and prints two fields. Any public JSON API can be used the same way.

Practical Example 4: An API that accepts a POST request to add a doctor

Replace doctor_list in views.py with a view that handles both GET and POST:
from rest_framework import status

@api_view(['GET', 'POST'])
def doctor_list(request):
    if request.method == 'GET':
        serializer = DoctorSerializer(Doctor.objects.all(), many=True)
        return Response(serializer.data)

    serializer = DoctorSerializer(data=request.data)       # deserialize the JSON body
    if serializer.is_valid():
        serializer.save()                                  # INSERT into the database
        return Response(serializer.data, status=status.HTTP_201_CREATED)
    return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

Practical Example 5: Class-based views for doctor CRUD

Replace the function view in views.py with two class-based views. DoctorList handles listing and creating; DoctorDetail handles reading, updating, and deleting one doctor.
from django.http import Http404
from rest_framework import status
from rest_framework.response import Response
from rest_framework.views import APIView
from .models import Doctor
from .serializers import DoctorSerializer

class DoctorList(APIView):
    def get(self, request):                                    # READ all
        serializer = DoctorSerializer(Doctor.objects.all(), many=True)
        return Response(serializer.data)

    def post(self, request):                                   # CREATE
        serializer = DoctorSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)              # sends 400 automatically on errors
        serializer.save()
        return Response(serializer.data, status=status.HTTP_201_CREATED)

class DoctorDetail(APIView):
    def get_object(self, pk):
        try:
            return Doctor.objects.get(pk=pk)
        except Doctor.DoesNotExist:
            raise Http404

    def get(self, request, pk):                                # READ one
        return Response(DoctorSerializer(self.get_object(pk)).data)

    def put(self, request, pk):                                # UPDATE (all fields)
        serializer = DoctorSerializer(self.get_object(pk), data=request.data)
        serializer.is_valid(raise_exception=True)
        serializer.save()
        return Response(serializer.data)

    def patch(self, request, pk):                              # UPDATE (some fields)
        serializer = DoctorSerializer(self.get_object(pk), data=request.data, partial=True)
        serializer.is_valid(raise_exception=True)
        serializer.save()
        return Response(serializer.data)

    def delete(self, request, pk):                             # DELETE
        self.get_object(pk).delete()
        return Response(status=status.HTTP_204_NO_CONTENT)

from rest_framework.pagination import PageNumberPagination

class DoctorPagination(PageNumberPagination):
    page_size = 5                           # default items per page
    page_size_query_param = 'page_size'     # client can send ?page_size=10
    max_page_size = 50                      # upper limit
Use it in the get method of DoctorList (views.py):
from .pagination import DoctorPagination

class DoctorList(APIView):
    def get(self, request):
        doctors = Doctor.objects.all()                       # Doctor.Meta.ordering keeps pages stable
        paginator = DoctorPagination()
        page = paginator.paginate_queryset(doctors, request, view=self)
        serializer = DoctorSerializer(page, many=True)
        return paginator.get_paginated_response(serializer.data)
    # post() stays the same as in Practical Example 5

Practical Example 11: RESTful API design with a ViewSet and router

A ModelViewSet provides list, create, retrieve, update, partial update, and delete in one class. Filters come from the URL query string.
views.py (add):
from rest_framework import viewsets

class DoctorViewSet(viewsets.ModelViewSet):
    serializer_class = DoctorSerializer

    def get_queryset(self):
        queryset = Doctor.objects.all()
        specialty = self.request.query_params.get('specialty')
        city = self.request.query_params.get('city')
        if specialty:
            queryset = queryset.filter(specialty__iexact=specialty)
        if city:
            queryset = queryset.filter(city__iexact=city)
        return queryset

views.py (add):
from django.contrib.auth import authenticate
from rest_framework.authtoken.models import Token
from rest_framework.decorators import api_view, permission_classes
from rest_framework.permissions import (AllowAny, IsAuthenticated,
                                        IsAuthenticatedOrReadOnly, IsAdminUser)
from .serializers import RegisterSerializer

@api_view(['POST'])
@permission_classes([AllowAny])
def register(request):
    serializer = RegisterSerializer(data=request.data)
    serializer.is_valid(raise_exception=True)
    user = serializer.save()
    token, _ = Token.objects.get_or_create(user=user)
    return Response({'user': serializer.data, 'token': token.key}, status=status.HTTP_201_CREATED)

@api_view(['POST'])
@permission_classes([AllowAny])
def login_view(request):
    user = authenticate(username=request.data.get('username'),
                        password=request.data.get('password'))
    if user is None:
        return Response({'error': 'Invalid username or password'}, status=status.HTTP_400_BAD_REQUEST)
    token, _ = Token.objects.get_or_create(user=user)
    return Response({'token': token.key, 'username': user.username})

@api_view(['POST'])
@permission_classes([IsAuthenticated])
def logout_view(request):
    request.user.auth_token.delete()                          # the old token stops working
    return Response({'message': 'Logged out successfully'})

@api_view(['GET'])
@permission_classes([IsAuthenticated])
def me(request):                                              # a protected endpoint
    return Response({'username': request.user.username, 'email': request.user.email})

Part B: OTP with Twilio
doctor_finder/services.py (new file):
import secrets
from datetime import timedelta

from django.conf import settings
from django.utils import timezone
from twilio.rest import Client

from .models import PhoneOTP

OTP_VALID_MINUTES = 5

def generate_otp():
    return str(secrets.randbelow(900000) + 100000)           # secure random 6-digit code

def send_otp_sms(phone):
    code = generate_otp()
    PhoneOTP.objects.filter(phone=phone).delete()            # keep only the latest OTP
    PhoneOTP.objects.create(phone=phone, code=code)
    client = Client(settings.TWILIO_ACCOUNT_SID, settings.TWILIO_AUTH_TOKEN)
    client.messages.create(
        body=f'Your Doctor Finder OTP is {code}. It is valid for {OTP_VALID_MINUTES} minutes.',
        from_=settings.TWILIO_PHONE_NUMBER,
        to=phone)                                            # E.164 format, e.g. +919876543210

def verify_otp(phone, code):
    otp = PhoneOTP.objects.filter(phone=phone, code=code, verified=False).first()
    if otp is None:
        return False
    if timezone.now() - otp.created_at > timedelta(minutes=OTP_VALID_MINUTES):
        otp.delete()                                         # expired
        return False
    otp.verified = True
    otp.save()
    return True
views.py (add):
from twilio.base.exceptions import TwilioRestException
from . import services

@api_view(['POST'])
@permission_classes([AllowAny])
def send_otp(request):
    phone = request.data.get('phone', '')
    if not phone.startswith('+') or not phone[1:].isdigit():
        return Response({'error': 'Phone must be in international format, e.g. +919876543210'},
                        status=status.HTTP_400_BAD_REQUEST)
    try:
        services.send_otp_sms(phone)
    except TwilioRestException as exc:
        return Response({'error': str(exc)}, status=status.HTTP_502_BAD_GATEWAY)
    return Response({'message': 'OTP sent'})

@api_view(['POST'])
@permission_classes([AllowAny])
def verify_otp_view(request):

import requests
from urllib.parse import quote

class ExternalAPIError(Exception):
    def __init__(self, message, status_code=502):
        super().__init__(message)
        self.status_code = status_code

def get_json(url, **kwargs):
    try:
        response = requests.get(url, timeout=10, **kwargs)
    except requests.RequestException as exc:
        raise ExternalAPIError(f'Network error: {exc}')
    if response.status_code == 404:
        raise ExternalAPIError('Not found', 404)
    if response.status_code == 429:
        raise ExternalAPIError('Rate limit reached, try again later', 429)
    if response.status_code in (401, 403):
        raise ExternalAPIError('The API rejected the key or token', 502)
    if not response.ok:
        raise ExternalAPIError(f'External API error {response.status_code}')
    return response.json()
views.py (add one small helper used by the views below):
def api_call(func, *args, **kwargs):
    try:
        return Response(func(*args, **kwargs))
    except ExternalAPIError as exc:
        return Response({'error': str(exc)}, status=exc.status_code)

import math

def haversine_km(lat1, lon1, lat2, lon2):
    r = 6371                                               # Earth radius in km
    p1, p2 = math.radians(lat1), math.radians(lat2)
    dphi = math.radians(lat2 - lat1)
    dlmb = math.radians(lon2 - lon1)
    a = math.sin(dphi / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dlmb / 2) ** 2
    return 2 * r * math.asin(math.sqrt(a))
