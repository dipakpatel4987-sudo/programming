# doctor/views.py  (add)
def profile_demo(request):
    doctor = {'name': 'Dr. Asha Shah', 'specialty': 'Cardiologist',
              'city': 'Ahmedabad', 'phone': '9876543210',
              'fee': 500, 'available': True}
    return render(request, 'doctor/profile_static.html', {'doctor': doctor})


# doctor/urls.py  (add inside urlpatterns)
    path('profile-demo/', views.profile_demo, name='profile-demo'),


<!-- doctor/templates/doctor/profile_static.html -->
