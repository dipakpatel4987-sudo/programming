Practical Example 12: CRUD operations on doctor profiles using the Django ORM

Form (doctor/forms.py, new file):
from django import forms
from .models import Doctor

class DoctorForm(forms.ModelForm):
    class Meta:
        model = Doctor
        fields = ['name', 'specialty', 'city', 'phone', 'email', 'fee',
                  'bio', 'available', 'latitude', 'longitude']

Views (doctor/views.py; this replaces the earlier doctor_list with a searchable version):

from django.db.models import Q
from django.shortcuts import render, get_object_or_404, redirect
from .forms import DoctorForm
from .models import Doctor

def doctor_list(request):                                   # READ (list + search)
    q = request.GET.get('q', '')
    doctors = Doctor.objects.all().order_by('name')
    if q:
        doctors = doctors.filter(Q(name__icontains=q) |
                                 Q(specialty__icontains=q) |
                                 Q(city__icontains=q))
    return render(request, 'doctor/doctor_list.html', {'doctors': doctors, 'q': q})

def doctor_create(request):                                 # CREATE
    form = DoctorForm(request.POST or None)
    if form.is_valid():
        form.save()
        return redirect('doctors')
    return render(request, 'doctor/doctor_form.html', {'form': form, 'title': 'Add Doctor'})

def doctor_update(request, id):                             # UPDATE
    doctor = get_object_or_404(Doctor, id=id)
    form = DoctorForm(request.POST or None, instance=doctor)
    if form.is_valid():
        form.save()
        return redirect('doctors')
    return render(request, 'doctor/doctor_form.html', {'form': form, 'title': 'Edit Doctor'})

def doctor_delete(request, id):                             # DELETE
    doctor = get_object_or_404(Doctor, id=id)
    if request.method == 'POST':
        doctor.delete()
        return redirect('doctors')
    return render(request, 'doctor/doctor_confirm_delete.html', {'doctor': doctor})

URLs (doctor/urls.py, add):
    path('doctors/add/', views.doctor_create, name='doctor-add'),
    path('doctors/<int:id>/edit/', views.doctor_update, name='doctor-edit'),
    path('doctors/<int:id>/delete/', views.doctor_delete, name='doctor-delete'),
