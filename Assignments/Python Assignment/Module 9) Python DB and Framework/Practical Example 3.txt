# doctor/views.py  (add)
def register_patient(request):
    message = ''
    if request.method == 'POST':
        message = 'Registration successful for ' + request.POST.get('name', '')
    return render(request, 'doctor/register_patient.html', {'message': message})

# doctor/urls.py  (add)
    path('register-patient/', views.register_patient, name='register-patient'),

<script>
document.getElementById('regForm').addEventListener('submit', function (e) {
    let valid = true;
    const v = id => document.getElementById(id).value.trim();
    const show = (id, msg) => { document.getElementById(id).textContent = msg; if (msg) valid = false; };

    show('nameErr',  v('name').length < 3 ? 'Name must be at least 3 characters' : '');
    show('emailErr', /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v('email')) ? '' : 'Enter a valid email');
    show('phoneErr', /^\d{10}$/.test(v('phone')) ? '' : 'Phone must be exactly 10 digits');
    show('ageErr',   (v('age') >= 1 && v('age') <= 120) ? '' : 'Age must be between 1 and 120');
    show('passErr',  v('password').length < 6 ? 'Password must be at least 6 characters' : '');
    show('confirmErr', v('password') !== v('confirm') ? 'Passwords do not match' : '');

    if (!valid) e.preventDefault();      // stop submission when any check fails
});
</script>
