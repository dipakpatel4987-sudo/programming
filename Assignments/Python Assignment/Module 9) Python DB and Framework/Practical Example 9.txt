Practical Example 9: URL routing (home, profile, contact)

Model for contact messages (doctor/models.py, add, then run makemigrations and migrate):
class ContactMessage(models.Model):
    name = models.CharField(max_length=100)
    email = models.EmailField()
    message = models.TextField()
    sent_at = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return f'{self.name}: {self.message[:30]}'

Views (doctor/views.py, add):

from django.shortcuts import get_object_or_404
from .models import ContactMessage

def doctor_profile(request, id):
    doctor = get_object_or_404(Doctor, id=id)
    return render(request, 'doctor/profile.html', {'doctor': doctor})

def contact(request):
    sent = False
    if request.method == 'POST':
        ContactMessage.objects.create(name=request.POST['name'],
                                      email=request.POST['email'],
                                      message=request.POST['message'])
        sent = True
    return render(request, 'doctor/contact.html', {'sent': sent})
